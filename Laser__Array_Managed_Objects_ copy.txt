class Laser extends Entity {
  float vx, vy;
  boolean active = false;

  Laser() {
    super(0, 0, 5, 5);
  }

  // Activates a recycled array slot without new object instantiation
  void spawn(float x, float y, float a) {
    this.x = x;
    this.y = y;
    this.vx = cos(a) * 5;
    this.vy = sin(a) * 5;
    this.active = true;
  }

  void update() {
    x += vx;
    y += vy;
    // Boundary check to deactivate lasers leaving screen bounds
    if (x < 0 || x > width || y < 0 || y > height) {
      active = false;
    }
  }

  void display() {
    noStroke();
    fill(255, 0, 0);
    ellipse(x, y, w, h);
  }
}
