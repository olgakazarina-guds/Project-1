import processing.sound.*;

// --- GLOBAL OBJECTS ---
Spaceship ship; 
Cannon cannon;
ArrayList<Meteorite> meteorites = new ArrayList<Meteorite>();
BossMeteorite boss = null;
Laser[] lasers = new Laser[10]; 
Star[] stars = new Star[200]; 

// --- AUDIO OBJECTS ---
SoundFile engineSnd;
SoundFile laserSnd;
SoundFile explosionSnd;

// --- GAME ASSETS & STATE ---
PImage meteorImg;
int gameState = 0; // 0=START, 1=PLAY, 2=GAMEOVER, 3=VICTORY
int score = 0;
int health = 100;
int shakeTimer = 0; 

void setup() { 
  size(600, 400); 
  
  // 1. Initialize Parallax Starfield
  for (int i = 0; i < stars.length; i++) {
    stars[i] = new Star(i % 3); 
  }
  
  // 2. Load Visual Assets
  meteorImg = loadImage("meteor.png"); 
  if (meteorImg == null) {
    meteorImg = createImage(60, 60, ARGB);
    meteorImg.loadPixels();
    for (int i = 0; i < meteorImg.pixels.length; i++) {
      meteorImg.pixels[i] = color(150, 150, 150);
    }
    meteorImg.updatePixels();
  }
  
  // 3. Initialize Audio
  try {
    engineSnd = new SoundFile(this, "engine.wav");
    laserSnd = new SoundFile(this, "laser.wav");
    explosionSnd = new SoundFile(this, "explosion.wav");
    engineSnd.loop();
    engineSnd.amp(0.1);
  } catch (Exception e) {
    println("Audio files missing in /data folder. Sound disabled.");
  }
  
  resetGame();
}

void resetGame() {
  score = 0;
  health = 100;
  shakeTimer = 0;
  boss = null;
  ship = new Spaceship(); 
  cannon = new Cannon(ship);
  
  meteorites.clear();
  for (int i = 0; i < 3; i++) {
    meteorites.add(new Meteorite(random(50, width - 50), random(-200, -50), meteorImg));
  }

  for (int i = 0; i < lasers.length; i++) { 
    lasers[i] = new Laser(); 
  } 
  
  // Restart engine sound if it was stopped
  if (engineSnd != null && !engineSnd.isPlaying()) {
    engineSnd.loop();
  }
}

void draw() { 
  background(10); 

  // Apply Screen Shake
  pushMatrix();
  if (shakeTimer > 0) {
    translate(random(-7, 7), random(-7, 7));
    shakeTimer--;
  }

  // State Machine
  if (gameState == 0) {
    drawStarfield();
    drawStartScreen();
  } else if (gameState == 1) {
    drawStarfield();
    drawGameLoop();
  } else if (gameState == 2) {
    if (engineSnd != null) engineSnd.stop();
    drawGameOverScreen();
  } else if (gameState == 3) {
    if (engineSnd != null) engineSnd.stop();
    drawVictoryScreen();
  }
  
  popMatrix();
}

// --- CORE GAMEPLAY LOOP ---

    float engineVol = map(mouseSpeed, 0, 40, 0.15, 0.4);
    float enginePan = map(ship.x, 0, width, -1.0, 1.0);
    
    engineSnd.rate(enginePitch);
    engineSnd.amp(engineVol);
    engineSnd.pan(enginePan);
  }

  ship.display(); 
  cannon.update();
  cannon.display();

  // Lasers logic
  for (int i = 0; i < lasers.length; i++) { 
    if (lasers[i].active) { 
      lasers[i].update(); 
      lasers[i].display();
    } 
  }

  if (score >= 150 && boss == null) {
    boss = new BossMeteorite(width / 2, -150, meteorImg);
  }

  if (boss != null) {
    boss.update();
    boss.display();
    
    for (int i = 0; i < lasers.length; i++) {
      if (lasers[i].active) {
        float d = dist(lasers[i].x, lasers[i].y, boss.x, boss.y);
        if (d < boss.w / 1.8) { 
          boss.hitEffect();
          if (explosionSnd != null) explosionSnd.play(0.6, 0.7); // Heavy blast
          shakeTimer = 6;
          lasers[i].active = false;
          if (boss.isDestroyed()) {
            score += 1000;
            gameState = 3; 
          }
        }
      }
    }
    if (boss.y > height) { health = 0; gameState = 2; }
  }

    Meteorite m = meteorites.get(j);

    m.display();

      if (lasers[i].active && dist(lasers[i].x, lasers[i].y, m.x, m.y) < m.w / 2) {
        m.hitEffect();
