Spaceship ship;
Meteorite meteor;

// FIX: Initialized array size to 10 to match design specification for Array Slot Reuse
Laser[] lasers = new Laser[10]; 

PImage meteorImg;

void setup() {
  size(600, 400); // Rule: size() must be the first line
  meteorImg = loadImage("meteor.png"); 
  ship = new Spaceship(); 
  meteor = new Meteorite(100, 100, meteorImg);

  for (int i = 0; i < lasers.length; i++) { 
    lasers[i] = new Laser(); 
  } 
}

void draw() {
  background(51); // Prevent "smearing"

  ship.update(); 
  ship.display(); 
  meteor.display();

  for (int i = 0; i < lasers.length; i++) { 
    if (lasers[i].active) { 
      lasers[i].update(); 
      lasers[i].display();
      
      // Collision detection between active lasers and the meteorite target
      float d = dist(lasers[i].x, lasers[i].y, meteor.x, meteor.y);
      if (d < meteor.w / 2) {
        meteor.hitEffect();       // Trigger pixel-level modification
        lasers[i].active = false; // Deactivate laser to immediately recycle the slot
      }
    } 
  } 
}

void mousePressed() { 
  // Find the first inactive laser slot in the array and recycle it
  for (int i = 0; i < lasers.length; i++) { 
    if (!lasers[i].active) { 
      lasers[i].spawn(ship.x, ship.y, ship.angle); 
      break; 
    } 
  } 
}
